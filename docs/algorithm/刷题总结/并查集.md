**547. 朋友圈**

使用邻接矩阵保存朋友关系，如果A[i][j]==1,是朋友关系，A->B->C->D的间接朋友关系也属于同一个朋友圈，求朋友圈个数。朋友关系的存储是典型的图（邻接矩阵），是一个二维数组。

使用BFS和DFS时，需要逐个遍历大数组里的小数组，并全局唯一保存是否访问的状态。

**方法1**：DFS。一条深搜路线就行一个朋友圈，当一个朋友圈的i, j匹配上时，才去更新全局唯一的visted状态j。起点i不需要加入visted状态，因为走完1次所有的j以后，以i为起点所有的路径都已经走过了，后续不会再有x->i这样的路线。

```java
//方法1：DFS
public int findCircleNum(int[][] M) &#123;
    if (M == null || M.length == 0) return 0;
    boolean[] visted = new boolean[M.length];
    int count = 0;
    for (int i = 0; i &lt; M.length; i++) &#123;
        if (!visted[i]) &#123;
            // visted[i] = true; 加上这行也能ac
            dfs(M, i, visted);
            //一条dfs线是一个朋友圈
            count++;       
        &#125;
    &#125;
    return count;
&#125;
private void dfs(int[][] M, int i, boolean[] visted) &#123;
    for (int j = 0; j &lt; M.length; j++) &#123;
        if (!visted[j] && M[i][j] == 1) &#123;
            visted[j] = true;
            dfs(M, j, visted);
        &#125;
    &#125;

public int findCircleNum(int[][] M) &#123;
    int n = M.length;
    //构建并查集
    QuickUnionUF uf = new QuickUnionUF(n);
    for (int i = 0; i &lt; n; i++) &#123;
        for (int j = 0; j &lt; n; j++) &#123;
            if (i != j && M[i][j] == 1) uf.union(i, j);
        &#125;
    &#125;
    //查询并查集的帮派数目
    int[] roots = uf.roots;
    int count = 0;
    for (int i = 0; i &lt; n; i++) &#123;
        if (roots[i] == i) count++;
    &#125;
    return count;
&#125;

private class QuickUnionUF &#123;

    private int[] roots;

    public QuickUnionUF(int n) &#123;
        roots = new int[n];
        for (int i = 0; i &lt; n; i++) &#123;
            roots[i] = i;
        &#125;
    &#125;

    private int findRoot(int i) &#123;
        int root = i;
        //找到老大root
        while (root != roots[root]) &#123;
            root = roots[root];
        &#125;
        //路径压缩
        while (i != roots[i]) &#123;
            int temp = roots[i];
            roots[i] = root;
            i = temp;
        &#125;
        return root;
    &#125;

    public boolean connected(int p, int q) &#123;
        return findRoot(p) == findRoot(q);
    &#125;

    public void union(int p, int q) &#123;
        int proot = findRoot(p);
        int qroot = findRoot(q);
        roots[proot] = qroot;
    &#125;
&#125;
```

**200. 岛屿数量**
