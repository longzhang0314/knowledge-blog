#### 翻译题意

**999. 车的可用捕获量**

题意是给定8 * 8的棋盘，自己是R，队友是B，敌人是p，R可以横冲直撞前进，遇到B停在B前面，遇到p吃掉p停在p的位置，遇到边缘停止。问p往上下左右四个方向走，能吃到p的方案有几种？

```java
//移动一次吃到卒的方案有多少种
public int numRookCaptures(char[][] board) &#123;
    int[][] directions = &#123;&#123;-1, 0&#125;, &#123;1, 0&#125;, &#123;0, -1&#125;, &#123;0, 1&#125;&#125;;

    for (int i = 0; i &lt; board.length; i++) &#123;
        for (int j = 0; j &lt; board[0].length; j++) &#123;
            //找到自己
            if (board[i][j] == 'R') &#123;
                int res = 0;
                //四种方案判断哪些合理
                for (int[] direction : directions) &#123;
                    if (burnout(board, i, j, direction)) &#123;
                        res++;
                    &#125;
                &#125;
                return res;
            &#125;
        &#125;
    &#125;
    return 0;
&#125;

//往一个方向走
private boolean burnout(char[][] board, int i, int j, int[] direction) &#123;
    
    while (inArea(i, j)) &#123;
        //遇到友军
        if (board[i][j] == 'B') break;
        //吃到卒直接返回
        if (board[i][j] == 'p') return true;
        //继续走一步
        i += direction[0];
        j += direction[1];
    &#125;
    return false;
&#125;

//是否没有到边上
private boolean inArea(int row, int col) &#123;
    return row >= 0 && row &lt; 8 && col >= 0 && col &lt; 8;
&#125;
```
