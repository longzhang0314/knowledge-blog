#### 并查集

并查集初步代码实现：

```java
public class QuickUnionUF &#123;

    private int[] roots;

    public QuickUnionUF(int n) &#123;
        roots = new int[n];
        for (int i = 0; i &lt; n; i++) &#123;
            roots[i] = i;
        &#125;
    &#125;

    private int findRoot(int i) &#123;
        int root = i;
        //找到老大root
        while (root != roots[root]) &#123;
            root = roots[root];
        &#125;
        //路径压缩
        while (i != roots[i]) &#123;
            int temp = roots[i];
            roots[i] = root;
            i = temp;
        &#125;
        return root;
    &#125;

    public boolean connected(int p, int q) &#123;
        return findRoot(p) == findRoot(q);
    &#125;

    public void union(int p, int q) &#123;
        int proot = findRoot(p);
        int qroot = findRoot(q);
        roots[proot] = qroot;
    &#125;

&#125;
```

进行rank优化，保证每次连接时小树连接到大树，并统计连通量个数。

```java
public class UnionQuickUF &#123;
    private int[] roots;
    private int[] rank;
    private int count;

    public UnionQuickUF(int n) &#123;
        this.roots = new int[n];
        this.rank = new int[n];
        for (int i = 0; i &lt; n; i++) &#123;
            roots[i] = i;
            rank[i] = 1;
        &#125;
    &#125;

    private int findRoot(int i) &#123;
        int root = i;
        //找到root节点
        while (root != roots[root]) &#123;
            root = roots[root];
        &#125;
        //路径压缩
        while (i != roots[root]) &#123;
            int temp = roots[i];
            //i直接指向root
            roots[i] = root;
            i = temp;
        &#125;
        return root;
    &#125;

    public boolean connected(int p, int q) &#123;
        return findRoot(p) == findRoot(q);
    &#125;
    

    
    public void union(int p, int q) &#123;
        int proot = findRoot(p);
        int qroot = findRoot(q);
        if (proot == qroot)
            return;
        //平衡性优化
        if (rank[proot] > rank[qroot]) &#123;
            roots[qroot] = proot;
            rank[proot] += rank[qroot];
        &#125; else &#123;
            roots[proot] = qroot;
            rank[qroot] += rank[proot];
        &#125;
        count--;
    &#125;
    
    public int count() &#123;
        return count;
    &#125;
&#125;
```
