#### 排序

- O(NlogN)

**1. 归并排序**

```java
//方法1:mergeSort
public int[] sortArray1(int[] nums) &#123;
    if (nums == null || nums.length &lt;= 1) return nums;
    mergeSort(nums, 0, nums.length - 1);
    return nums;

private void mergeSort(int[] nums, int p, int r) &#123;
    if (p >= r) return;
    int q = (p + r) >>> 1;
    mergeSort(nums, p, q);
    mergeSort(nums, q + 1, r);
    merge(nums, p, q, r);
&#125;

private void merge(int[] nums, int p, int q, int r) &#123;
    int[] temp = new int[r - p + 1];
    int i = p;
    int j = q + 1;
    int k = 0;
    while (i &lt;= q && j &lt;= r) &#123;
        if (nums[i] &lt; nums[j]) temp[k++] = nums[i++];
        else temp[k++] = nums[j++];
    &#125;

    int start = i;
    int end = q;
    if (j &lt;= r) &#123;
        start = j;
        end = r;
    &#125;

    while (start &lt;= end) &#123;
        temp[k++] = nums[start++];
    &#125;

    for (i = 0; i &lt; k; i++) &#123;
        nums[p + i] = temp[i];
    &#125;
&#125;
```

**2. 快排**

```java
//方法2：quickSort
public int[] sortArray(int[] nums) &#123;
    if (nums == null || nums.length &lt;= 1) return nums;
    quickSort(nums, 0, nums.length - 1);
    return nums;
&#125;

private void quickSort(int[] nums, int p, int r) &#123;
    if (p >= r) return;
    int patition = patition(nums, p, r);
    quickSort(nums, p, patition - 1);
    quickSort(nums, patition + 1, r);
&#125;

private int patition(int[] nums, int p, int r) &#123;
    int pivot = nums[r];
    int j = p;
    for (int i = p; i &lt; r; i++) &#123;
        if (nums[i] &lt;= pivot) &#123;
            //说明i走的比j多，此时j停留位置大于pivot
            if (i != j) &#123;
                swap(nums, i, j);
            &#125;
            j++;
        &#125;
    &#125;
    //此时j停留的位置比pivot大，交换位置
    swap(nums, j, r);
    //返回分区点
    return j;
&#125;

private void swap(int[] nums, int i, int j) &#123;
    int temp = nums[i];
    nums[i] = nums[j];
    nums[j] = temp;
&#125;
```

**5. 计数排序**

```java
//方法5: 计数排序
//1 &lt;= A.length &lt;= 10000
//-50000 &lt;= A[i] &lt;= 50000
private static final int ADDITION_CAPACITY = 50000;
public int[] sortArray(int[] nums) &#123;
    if (nums == null) return nums;
    int n = nums.length;
    if (n &lt;= 1) return nums;
    int max = Integer.MIN_VALUE;
    for (int num : nums) &#123;
        max = Math.max(max, num);
    &#125;
    int[] counter = new int[max + 1 + ADDITION_CAPACITY];
    for (int num : nums) &#123;
        counter[num + ADDITION_CAPACITY]++;
    &#125;
    for (int i = 1; i &lt; counter.length; i++) &#123;
        counter[i] += counter[i - 1];
    &#125;
    int[] temp = new int[n];
    //倒序的目的是如果有相同的值，不改变原顺序
    for (int i = n - 1; i >= 0; i--) &#123;
        //小于等于nums[i] + ADDITION_CAPACITY的最后一个索引
        int index = --counter[nums[i] + ADDITION_CAPACITY];
        temp[index] = nums[i];
    &#125;
    return temp;
&#125;
```
